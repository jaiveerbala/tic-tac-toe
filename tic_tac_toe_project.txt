# Gameplay

def ticTacToe():
  
  coinflip = random.randint(0, 1)
  
  if coinflip == 0:
    
    print("The coin flip shows that you (X) will go first!")
    
  elif coinflip == 1:
    
    print("The coin flip shows that the computer (O) will go first!")
  
  gslPrinter()
    
  while True:
    
    if coinflip == 0:
      
      print("")
      
      userrow = int(input("Pick a row to play: ")) - 1
      usercol = int(input("Pick a column to play: ")) - 1
      
      print("")
      
      gslUpdaterPlayer(userrow, usercol)
      
    elif coinflip == 1:
      
      print("")
      
      print("The computer has made its move")
      
      print("")
      
      gslUpdaterComputer()
    
    gslPrinter()
      
    if wonGame() == 1:
      
      print("You won! Congratualtions!")
      break
    
    elif wonGame() == 2:
      
      print("The computer won! Try again!")
      break
    
    elif wonGame() == 3:
      
      print("Tie! Play again to decide the true champion!")
      break
    
    if coinflip == 0:
      
      coinflip = 1
      
    elif coinflip == 1:
      
      coinflip = 0
      
ticTacToe()
